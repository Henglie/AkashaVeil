    set(index 0)
    foreach(arch IN ITEMS x86 x64 arm64)
        math(EXPR index "${index}+1")
        set(worker "${AV_PAYLOAD_DIR}/${arch}/AkashaVeilWorker.exe")
        set(hook "${AV_PAYLOAD_DIR}/${arch}/AkashaVeilHook-${arch}.dll")
        foreach(input IN ITEMS "${worker}" "${hook}")
            if(NOT EXISTS "${input}")
                message(FATAL_ERROR "Missing embedded payload. Run tools/build.ps1 first: ${input}")
            endif()
            file(SHA256 "${input}" hash)
            string(APPEND bundle_hashes "${hash}")
            list(APPEND bundle_files "${input}")
        endforeach()
        math(EXPR worker_id "100+${index}")
        math(EXPR hook_id "200+${index}")
        string(APPEND bundle_rc "${worker_id} RCDATA \"${worker}\"\n${hook_id} RCDATA \"${hook}\"\n")
    endforeach()
    string(SHA256 AV_PAYLOAD_ID "${bundle_hashes}")
    configure_file(cmake/payload_manifest.h.in generated/payload_manifest.h @ONLY)
    file(WRITE "${CMAKE_BINARY_DIR}/generated/payloads.rc" "${bundle_rc}")
